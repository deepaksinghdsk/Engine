cmake_minimum_required(VERSION 3.10)
project(VulkanEngine VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2")

# Fix PDB locking issues with parallel builds on MSVC
if(MSVC)
    string(APPEND CMAKE_CXX_FLAGS " /FS")
endif()

include(CTEST)
enable_testing()

find_package(Vulkan REQUIRED)

# include paths
set(GLFW_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/dependencies/GLFW/include/")
#set(VULKAN_INCLUDE_DIR "D:/VulkanSDK/1.3.268.0/Include/")
set(GLFW_LIB_PATH "${CMAKE_SOURCE_DIR}/dependencies/GLFW/lib-vc2022/glfw3.lib")
#message(STATUS "GLFW LIB path: ${GLFW_LIB_PATH}")

set(GLM_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/src/thirdParty/glm-0.9.9.8/")
set(THIRDPARTY_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/src/thirdParty/")

set(IMGUI_INCUDE_DIR "${CMAKE_SOURCE_DIR}/src/thirdParty/imgui/")
include_directories(${IMGUI_INCUDE_DIR} ${IMGUI_INCUDE_DIR}/backends)

add_definitions(-DGLFW_STATIC)

option(GLFW_BUILD_EXAMPLES "Build the GLFW example programs" OFF)
option(GLFW_BUILD_TESTS "Build the GLFW test programs" OFF)
option(GLFW_BUILD_DOCS "Build the GLFW documentation" OFF)
option(GLFW_INSTALL "Generate installation target" OFF)
option(GLFW_DOCUMENT_INTERNALS "Include internals in documentation" OFF)
include_directories(${GLFW_INCLUDE_DIR})

#include_directories(${VULKAN_INCLUDE_DIR})

include_directories(${GLM_INCLUDE_DIR})
include_directories(${THIRDPARTY_INCLUDE_DIR}) 

set(VULKAN_ENGINE_SOURCES
                    app/src/Bloom.cpp
                    src/Backend/Application.cpp
                    src/Backend/Context.cpp
                    src/Backend/VulkanSwapChain.cpp
                    src/Backend/VulkanDevice.cpp
                    src/Backend/Pipeline.cpp
                    src/Backend/CommandBuffer.cpp
                    src/Backend/Buffer.cpp
                    src/Backend/RenderPass.cpp
                    src/Backend/DescriptorManager.cpp
                    src/Backend/Camera.h
                    src/Backend/Image.cpp
                    src/Backend/Model.cpp    
)

set(IMGUI_SOURCES
        src/thirdParty/imgui/backends/imgui_impl_glfw.cpp 
        src/thirdParty/imgui/backends/imgui_impl_vulkan.cpp 
        src/thirdParty/imgui/imgui.cpp 
        src/thirdParty/imgui/imgui_draw.cpp
        src/thirdParty/imgui/imgui_demo.cpp 
        src/thirdParty/imgui/imgui_tables.cpp 
        src/thirdParty/imgui/imgui_widgets.cpp
)

add_executable(VulkanEngine 
                    ${VULKAN_ENGINE_SOURCES}
                    ${IMGUI_SOURCES}
)

target_include_directories(VulkanEngine PRIVATE
                    ${CMAKE_CURRENT_SOURCE_DIR}/src/
                    ${CMAKE_CURRENT_SOURCE_DIR}/app/src/
) 

target_link_libraries(VulkanEngine
    Vulkan::Vulkan
    ${GLFW_LIB_PATH}
)

# Fix linker warning about MSVCRT
if(MSVC)
    set_target_properties(VulkanEngine PROPERTIES
        LINK_FLAGS "/NODEFAULTLIB:MSVCRT"
    )
endif()

# Add test target
add_custom_target(test1
    COMMAND VulkanTest
    DEPENDS VulkanTest
    COMMENT "Running VulkanTest"
)

# Add custom clean
add_custom_target(clean-custom
    COMMAND ${CMAKE_COMMAND} -E remove VulkanTest.exe
    COMMENT "Cleaning VulkanTest binary"
)
